// ------------------------------------------------------------
// 	A sniper rifle, chambered in 30.06.
//	The bolt action has built-in dirt protection, so it doesn't need cleaning.
//	In return, the bolt is already stiff on it's own and lower strength players
//	may struggle working it.	--- Slogstin
// ------------------------------------------------------------

class VirStingray:HDWeapon{
	default{
		weapon.selectionorder 31;
		weapon.slotnumber 4;
		weapon.slotpriority 1;
		weapon.bobrangex 0.21;
		weapon.bobrangey 0.9;
		scale 0.7;
		inventory.pickupmessage "$PICKUP_VIRIDSTINGRAY";
		hdweapon.barrelsize 40,1,2;
		hdweapon.refid "SBS";
		tag "$TAG_VIRIDSTINGRAY";
		obituary "$OB_VIRIDSTINGRAY";
		inventory.icon "VBOWA0";
	}
	
	override bool AddSpareWeapon(actor newowner){return AddSpareWeaponRegular(newowner);}
	override hdweapon GetSpareWeapon(actor newowner,bool reverse,bool doselect){return GetSpareWeaponRegular(newowner,reverse,doselect);}
	
	override double weaponbulk(){
		return 70;
	}
	override double gunmass(){
		return 9;
	}
	
	override string gethelptext(){
		return
		WEPHELP_FIRE.."  Fire\n"
		..WEPHELP_ALTFIRE.."   Cycle bolt\n"
		..WEPHELP_RELOAD.."   Load Magazine\n"
		..WEPHELP_UNLOAD.."   Unload Magazine\n"
		;
	}
	
	override string,double getpickupsprite(){return "VBOWA0",1.;}
	
	override void ForceBasicAmmo(){
		owner.A_TakeInventory("Savage300Ammo");
		owner.A_GiveInventory("Savage300Ammo");
		ForceOneBasicAmmo("Vir_StingrayMag");
	}
	
	
		states{
	
	// ----- SELECT -----
	
	select0:
		BARG A 0 ;
		#### # 0;
		goto select0big;
	deselect0:
		BARG A 0 A_JumpIf(invoker.weaponstatus[VBOW_LOADED]==0,4);
		#### A 0 ;
		#### A 3;
		#### # 0;
		goto deselect0big;
	ready:
		#### # 1;
//		#### # 0 A_JumpIf(pressingreload(),"reload");
		#### # 0 A_WeaponReady();
		goto readyend;
		
	// ----- RELOAD -----
	
	reload:
	magout:
	loadmag:
		
	// ----- FIRE -----
		
	fire:
	shoot:
		#### # 0 A_JumpIf(!invoker.weaponstatus[VRAY_CHAMBER]==2,"nope");
	flash:
		#### # 0 A_StartSound("weapons/savage_shot",CHAN_WEAPON);
		BARG F 1 {
			        let bbb=HDBulletActor.FireBullet(self,"HDB_300Savage",spread:1.,speedfactor:frandom(1.3,1.35));
					invoker.weaponstatus[VRAY_CHAMBER]=1;
		}
		#### # 0;
		goto nope;
		
		
	// ----- ALTFIRE -----
	
	altfire:
	pullbolt:
		BARG A 1;
	holdbolt:
	releasebolt:
		
	// ----- SPAWN -----
	
	spawn:
		VBOW A -1;
		stop;
	}
}

enum vstingraystatus{
	VRAY_CHAMBER=2,
	VRAY_MAG=1,
};
